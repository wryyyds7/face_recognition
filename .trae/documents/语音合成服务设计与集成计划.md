#### 2. Feign客户端设计

**创建Feign客户端接口**：

```java
package com.homework.common.feign;

import org.springframework.cloud.openfeign.FeignClient;
import org.springframework.web.bind.annotation.PostMapping;
import org.springframework.web.bind.annotation.RequestParam;

/**
 * 语音合成Feign客户端
 */
@FeignClient(name = "voice-synthesis", fallbackFactory = VoiceSynthesisFallbackFactory.class)
public interface VoiceSynthesisClient {
    
    /**
     * 语音合成并播放
     * @param text 要合成的文本
     * @param voiceType 音色类型
     * @return 执行结果
     */
    @PostMapping("/voice/speak")
    boolean speak(@RequestParam("text") String text,
                 @RequestParam(value = "voiceType", defaultValue = "female") String voiceType);
}
```

**创建降级工厂**：

```java
package com.homework.common.feign.fallback;

import com.homework.common.feign.VoiceSynthesisClient;
import feign.hystrix.FallbackFactory;
import org.slf4j.Logger;
import org.slf4j.LoggerFactory;
import org.springframework.stereotype.Component;

@Component
public class VoiceSynthesisFallbackFactory implements FallbackFactory<VoiceSynthesisClient> {
    
    private static final Logger log = LoggerFactory.getLogger(VoiceSynthesisFallbackFactory.class);
    
    @Override
    public VoiceSynthesisClient create(Throwable throwable) {
        log.error("语音合成服务调用失败：{}", throwable.getMessage());
        return (text, voiceType) -> {
            log.error("语音合成失败，无法播放：{}", text);
            return false;
        };
    }
}
```

#### 3. 语音合成服务实现

**服务接口**：

```java
package com.homework.voicesynthesis.service;

/**
 * 语音合成服务接口
 */
public interface VoiceService {
    
    /**
     * 语音合成并播放
     * @param text 要合成的文本
     * @param voiceType 音色类型
     * @return 执行结果
     */
    boolean speak(String text, String voiceType);
    
    /**
     * 语音合成并播放（使用默认音色）
     * @param text 要合成的文本
     * @return 执行结果
     */
    boolean speak(String text);
}
```

**服务实现**：

```java
package com.homework.voicesynthesis.service.impl;

import com.homework.voicesynthesis.service.VoiceService;
import com.homework.voicesynthesis.config.VoiceConfig;
import com.jacob.activeX.ActiveXComponent;
import com.jacob.com.Dispatch;
import org.slf4j.Logger;
import org.slf4j.LoggerFactory;
import org.springframework.beans.factory.annotation.Autowired;
import org.springframework.stereotype.Service;

/**
 * 语音合成服务实现
 */
@Service
public class VoiceServiceImpl implements VoiceService {
    
    private static final Logger log = LoggerFactory.getLogger(VoiceServiceImpl.class);
    
    private final VoiceConfig voiceConfig;
    
    @Autowired
    public VoiceServiceImpl(VoiceConfig voiceConfig) {
        this.voiceConfig = voiceConfig;
    }
    
    @Override
    public boolean speak(String text, String voiceType) {
        try {
            // 使用Jacob库调用Windows TTS
            ActiveXComponent sap = new ActiveXComponent("Sapi.SpVoice");
            Dispatch sapo = sap.getObject();
            
            // 设置音量（0-100）
            Dispatch.put(sapo, "Volume", new Variant(voiceConfig.getVolume()));
            
            // 设置语速（-10到+10）
            Dispatch.put(sapo, "Rate", new Variant(voiceConfig.getRate()));
            
            // 选择音色
            Dispatch voices = Dispatch.call(sapo, "GetVoices").toDispatch();
            int voiceIndex = "male".equalsIgnoreCase(voiceType) ? 1 : 0;
            Dispatch voice = Dispatch.call(voices, "Item", new Variant(voiceIndex)).toDispatch();
            Dispatch.put(sapo, "Voice", voice);
            
            // 执行语音合成
            Dispatch.call(sapo, "Speak", new Variant(text));
            
            // 释放资源
            sapo.safeRelease();
            sap.safeRelease();
            
            log.info("语音合成成功：{}", text);
            return true;
        } catch (Exception e) {
            log.error("语音合成失败：{}", e.getMessage(), e);
            return false;
        }
    }
    
    @Override
    public boolean speak(String text) {
        return speak(text, voiceConfig.getDefaultVoiceType());
    }
}
```
